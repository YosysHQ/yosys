# every pattern it holds, so folding this in would make -combine/-expand pay
# to index every $add in the design on invocations that never use it.

	peepopt_sink_shifts.pmg
	PREFIX
		peepopt_sink
)

	opt_shift.cc
	${PMGEN_peepopt_shift_OUTPUT}

)
